from pathlib import Path


def master_key(settings, value):
    path = settings.master_key_path
    path.write_bytes(value)


def key_dir(cfg, name, v):
    p = Path(cfg.key_dir) / name
    p.write_text(v)


def public_report(target, v):
    p = Path(target)
    p.write_text(v)
