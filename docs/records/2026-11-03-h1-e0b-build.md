```python
"""Generate tests/fixtures/h1/definition_owners.json from main (blob-identical sources)."""
import ast, json, subprocess, sys

D = "src/decoy_engine/"
SLICES = {  # slice: (from file, [(first line, last line or name, planned owner)])
    "E1.1": ("detection/detectors.py", [(84, 205, "detection/_scoring.py")]),
    "E1.2": ("detection/detectors.py", [(216, 325, "detection/_hints.py")]),
    "E1.3": ("detection/detectors.py", [(337, 735, "detection/builtin.py")]),
    "E2": ("plan/_checks.py", [(287, 405, "plan/_checks_statistical.py"),
                               (406, 470, "plan/_checks_ner.py"),
                               (471, 580, "plan/_checks_fpe_checksum.py")]),
    "E3": ("profiling/eval/corpus.py", [(48, 286, "profiling/eval/_corpus_base.py"),
                                        (295, 597, "profiling/eval/_corpus_values.py")]),
    "E6a": ("generation/synthesize.py", [(n, n, "generation/_synthesize_derived.py")
                                         for n in (345, 518, 556)]),
}


def definitions(source):
    """(first line, name) of each top-level def, class, Assign target or valued AnnAssign."""
    for s in ast.parse(source).body:
        first = min([s.lineno, *(d.lineno for d in getattr(s, "decorator_list", []))])
        if isinstance(s, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
            yield first, s.name
        elif isinstance(s, (ast.Assign, ast.AnnAssign)) and s.value is not None:
            targets = s.targets if isinstance(s, ast.Assign) else [s.target]
            for t in targets:
                for n in ast.walk(t):
                    if isinstance(n, ast.Name) and isinstance(n.ctx, ast.Store):
                        yield first, n.id


rev = sys.argv[1]
out, moved = {}, {}
for slice_id, (src, ranges) in SLICES.items():
    text = subprocess.run(["git", "show", f"{rev}:{D}{src}"], capture_output=True, text=True, check=True).stdout
    owners = {name: D + owner for line, name in definitions(text)
              for lo, hi, owner in ranges if lo <= line <= hi}
    out[slice_id] = {"landed": False, "from": D + src, "owners": owners}
    moved.setdefault(D + src, set()).update(owners)
stays = {}
for path, names in moved.items():
    text = subprocess.run(["git", "show", f"{rev}:{path}"], capture_output=True, text=True, check=True).stdout
    stays[path] = sorted(dict.fromkeys(n for _, n in definitions(text) if n not in names))
out["stays"] = stays
print(json.dumps(out, indent=1))
```
