## 2 Table generation

`dump_ucd.py` ran once under each interpreter; `gen_table.py` ran under system Python with `-I` over both dumps.

```python
# dump_ucd.py <out.json>
import json, sys, unicodedata
rows = [[cp, unicodedata.category(chr(cp)), unicodedata.combining(chr(cp)), unicodedata.decomposition(chr(cp))] for cp in range(0x110000)]
json.dump({"unidata_version": unicodedata.unidata_version, "rows": rows}, open(sys.argv[1], "w"))

# gen_table.py <ucd310.json> <ucd312.json>
import hashlib, json, sys
old = json.load(open(sys.argv[1])); new = json.load(open(sys.argv[2]))
assert old["unidata_version"] == "13.0.0" and new["unidata_version"] == "15.0.0"
added = [n for o, n in zip(old["rows"], new["rows"]) if o[1] == "Cn" and n[1] != "Cn"]
table = sorted(n[0] for n in added if n[2] != 0 or n[3] != "")
text = ",".join(f"{cp:04X}" for cp in table)
print(len(added), len(table), hashlib.sha256(text.encode("ascii")).hexdigest())
```
